export function KpiCalculator() {
  const { department } = useParams()
  const { data } = useKpiScores(department)

  return (
    <div className="grid gap-4">
      {data.metrics.map(metric => (
        <ScoreCard
          key={metric.id}
          name={metric.name}
          weight={metric.weight}
          achievement={metric.value}
        />
      ))}
      <TotalScore score={data.total} />
    </div>
  )
}
